# 开手电筒（只支持骁龙设备）

# 关手电筒（只支持骁龙设备）
echo torch0_trigger > /sys/class/leds/led:torch_0/trigger;echo 0 > /sys/class/leds/led:torch_0/brightness;echo switch0_trigger > /sys/class/leds/led:switch_0/trigger;echo torch0_trigger > /sys/class/leds/led:torch_0/trigger

# 开手电筒（只支持天玑设备，在一加Ace5至尊版上通过测试，其他设备路径可能略有不同）
echo 4 > /sys/class/leds/white:flash-1/brightness;echo 4 > /sys/class/leds/white:flash-3/brightness

# 关手电筒（只支持天玑设备，在一加Ace5至尊版上通过测试，其他设备路径可能略有不同）
echo 0 > /sys/class/leds/white:flash-1/brightness;echo 0 > /sys/class/leds/white:flash-3/brightness
